  logScale(or){ const lo=Math.log(0.6), hi=Math.log(3.0); return ((Math.log(or)-lo)/(hi-lo))*100; }
  // two-sided p-value, one-proportion z-test vs baseline p0 (Zelen-Severo normal tail approx)
  pTwoSided(phat, p0, n){
    const se=Math.sqrt(p0*(1-p0)/n);
    const z=Math.abs((phat-p0)/se);
    const t=1/(1+0.2316419*z), d=0.3989423*Math.exp(-z*z/2);
    const upper=d*t*(0.3193815+t*(-0.3565638+t*(1.781478+t*(-1.821256+t*1.330274))));
    return 2*upper;
  }

    };
    const ageRaw = [

      { h:'Conclusions', body:'The models surfaced risk factors that standard regression failed to identify, yet individual readmission remains hard to predict — feature discovery is necessary but not sufficient for a useful risk measure.     ];
    const RB={ sig:'rgba(111,178,240,.08)', sigB:'rgba(111,178,240,.22)', null:'rgba(255,255,255,.03)', nullB:'var(--line)', warn:'rgba(216,176,106,.07)', warnB:'rgba(216,176,106,.24)' };
